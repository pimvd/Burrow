# Games Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a third child tab holding small built-in SwiftUI games, switched on per profile by the parent.

**Architecture:** Games are a fixed catalog in code (like `AvatarCatalog`), not SwiftData rows. The only per-profile state is `ChildProfile.enabledGameIDs`. Each game carries a stable hardcoded UUID and stands in as its own pseudo-tile in usage rows, so `StatisticsService` needs no changes at all.

**Tech Stack:** SwiftUI, SwiftData, Swift Testing (`import Testing`), Xcode 16, iOS 17+.

**Spec:** `docs/superpowers/specs/2026-08-16-games-tab-design.md`

## Global Constraints

- iOS deployment target is **17.0**. Swift language version **5.0**.
- The only third-party dependency is ZIPFoundation. **Do not add dependencies.**
- Every new SwiftData property must have a **default value or be optional**, with no uniqueness constraints — the store is mirrored to CloudKit.
- Every user-facing string uses `String(localized:comment:)` or `LocalizedStringResource` and must be added to `Kidtopia/Resources/Localizable.xcstrings` in **en, nl, de, fr, es**.
- The child interface has **no sound effects**. Games add none.
- Animations use `CalmStyle.transition`. Buttons use `.buttonStyle(.calm)`. Cards use `.calmCard(cornerRadius:)`.
- Games have **no age metadata**. Availability is the parent's toggle and nothing else. `AgeLayoutScale` is used only for visual sizing.
- New files go in existing folders; the `Kidtopia` group is a `PBXFileSystemSynchronizedRootGroup`, so **files on disk are picked up automatically** — no pbxproj edit is needed for new source files (Task 1 is the sole exception, which adds a whole target).
- Build check used throughout:
  `xcodebuild build -project Kidtopia.xcodeproj -scheme Kidtopia -configuration Debug -destination 'generic/platform=iOS Simulator' CODE_SIGNING_ALLOWED=NO CODE_SIGN_ENTITLEMENTS=""`

```swift
import Foundation

/// The fixed set of games that ship with the app, in the shape of
/// `AvatarCatalog` and `ThemePalette`.
///
/// A game is code, not content: it is identical on every device and there is
/// nothing about it to sync. Only *which* games a profile may see is stored,
/// on `ChildProfile.enabledGameIDs`.
enum GameCatalog {
    struct Game: Identifiable, Hashable, Sendable {
        /// Stable string stored in `ChildProfile.enabledGameIDs`.
        let identifier: String

        /// A stable, hardcoded UUID. Usage rows use it in place of a tile ID,
        /// which is what lets `StatisticsService` treat a game exactly like a
        /// tile without knowing games exist. Never generate this at runtime:
        /// a fresh UUID would orphan every row written by an earlier launch.
        let id: UUID

        let symbolName: String
        let titleResource: LocalizedStringResource

        var localizedTitle: String { String(localized: titleResource) }
    }

    static let matchingPairs = "matchingPairs"

    static let all: [Game] = [
        Game(
            identifier: matchingPairs,
            id: UUID(uuidString: "9F1B7A2C-4E63-4C1E-9D5A-2B8F0C6D3A11")!,
            symbolName: "square.grid.2x2",
            titleResource: LocalizedStringResource("Matching Pairs", comment: "Game name")
        )
    ]

    static func game(identifier: String) -> Game? {
        all.first { $0.identifier == identifier }
    }

    static func game(id: UUID) -> Game? {
        all.first { $0.id == id }
    }
}
```

    ///
    /// A game has no playlist, so the visit is both the opening and the single
    /// entry: the game's stable UUID stands in for the tile *and* for the
    /// media item. That is what lets `StatisticsService` sum game time without
    /// knowing games exist — a play row is recognised by a non-nil

    func beginGame(
        profileID: UUID,
        game: GameCatalog.Game,
        in context: ModelContext,
        now: Date = Date()

        finishCurrentEntry(in: context, now: now)
        self.profileID = profileID
        self.tileID = game.id
        self.gameIdentifier = game.identifier

        let marker = UsageSession(
            profileID: profileID,
            tileID: game.id,
            mediaItemID: nil,
            gameIdentifier: game.identifier,
            startTime: now,
            durationSeconds: 0
        )
        context.insert(marker)
        ModelSaver.save(context, describing: "recording a game opening")

        currentEntryID = game.id
        currentEntryStart = now
    }

    /// Ends the whole visit to a game.
    func endGame(in context: ModelContext, now: Date = Date()) {
        endTile(in: context, now: now)

```swift
import Foundation

/// The rules of Matching Pairs, with no SwiftUI in sight so they can be
/// tested directly.
///
/// There is no score, no timer and no way to lose: the board simply completes.
struct MatchingPairsBoard: Equatable {
    struct Card: Identifiable, Equatable {
        let id: UUID
        let symbolName: String
        var isFaceUp = false
        var isMatched = false
    }

    enum Outcome: Equatable {
        case ignored
        case revealed
        case matched
        case mismatched
    }

    private(set) var cards: [Card]

    /// `true` between a mismatch and the moment the two cards turn back down.
    /// Further taps are ignored while it holds, so a child cannot flip a third
    /// card into a half-resolved pair.
    private(set) var isResolvingMismatch = false

    var isComplete: Bool { cards.allSatisfy(\.isMatched) }

    /// Two cards per symbol, in the order given. Shuffling is the caller's job
    /// so tests can address specific cards.
    init(symbols: [String]) {
        cards = symbols.flatMap { symbol in
            [Card(id: UUID(), symbolName: symbol), Card(id: UUID(), symbolName: symbol)]
        }
    }

    mutating func shuffle() {
        cards.shuffle()
    }

    @discardableResult
    mutating func choose(_ id: UUID) -> Outcome {
        guard !isResolvingMismatch,
              let index = cards.firstIndex(where: { $0.id == id }),
              !cards[index].isMatched,
              !cards[index].isFaceUp else {
            return .ignored
        }

        cards[index].isFaceUp = true

        let faceUp = cards.indices.filter { cards[$0].isFaceUp && !cards[$0].isMatched }
        guard faceUp.count == 2 else { return .revealed }

        if cards[faceUp[0]].symbolName == cards[faceUp[1]].symbolName {
            cards[faceUp[0]].isMatched = true
            cards[faceUp[1]].isMatched = true
            return .matched
        }

        isResolvingMismatch = true
        return .mismatched
    }

    /// Turns the two mismatched cards back down and unblocks the board.
    mutating func hideMismatched() {
        for index in cards.indices where !cards[index].isMatched {
            cards[index].isFaceUp = false
        }
        isResolvingMismatch = false
    }
}
```

/// Turn cards over two at a time and find the pairs. No sound, no score and
/// no timer — it simply finishes and offers another round.

    /// Six pairs for older children, four for the youngest, so a board stays
    /// finishable. This is layout comfort, not an availability rule.

    ]

    private static let mismatchLinger = Duration.milliseconds(900)

    @State private var board = MatchingPairsBoard(symbols: [])
    @State private var mismatchToken = 0

    private var scale: AgeLayoutScale { profile.layoutScale }

    private var pairCount: Int { scale.ageYears <= 4 ? 4 : 6 }

    var body: some View {

            LazyVGrid(
                columns: [GridItem(.adaptive(minimum: 90), spacing: 16)],
                spacing: 16
            ) {
                ForEach(board.cards) { card in
                    Button {
                        reveal(card)
                    } label: {
                        cardFace(card)
                    }
                    .buttonStyle(.calm)
                    .disabled(card.isMatched || board.isResolvingMismatch)
                    .accessibilityLabel(
                        card.isFaceUp
                            ? Text("Card showing \(card.symbolName)", comment: "Face up game card")
                            : Text("Face down card", comment: "Face down game card")
                    )
                }
            }
            .padding(.horizontal, 20)

            if board.isComplete {
                Button {
                    startRound()
                } label: {
                    Text("Play again", comment: "Restart a game")
                        .font(.title3.weight(.semibold))
                        .padding(.horizontal, 28)
                        .padding(.vertical, 14)
                        .background(.thinMaterial, in: Capsule())
                }
                .buttonStyle(.calm)
                .transition(.opacity)
            }
        }
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .animation(CalmStyle.transition, value: board)
        .onAppear(perform: startRound)
        .task(id: mismatchToken) {
            guard board.isResolvingMismatch else { return }
            try? await Task.sleep(for: Self.mismatchLinger)
            guard !Task.isCancelled else { return }
            board.hideMismatched()
        }
    }

    @ViewBuilder
    private func cardFace(_ card: MatchingPairsBoard.Card) -> some View {
        ZStack {
            RoundedRectangle(cornerRadius: 20, style: .continuous)
                .fill(card.isFaceUp ? profile.theme.wash : Color(.secondarySystemBackground))

            if card.isFaceUp {
                Image(systemName: card.symbolName)
                    .font(.system(size: 40, weight: .light))
                    .foregroundStyle(profile.theme.tint)
            }

        board = fresh

            case dailyScreenTimeLimitMinutes, lockScreenPlaybackEnabled
            case visibleSharedTileIDs, enabledGameIDs
        }

        init(from decoder: any Decoder) throws {
            let container = try decoder.container(keyedBy: CodingKeys.self)
            id = try container.decode(UUID.self, forKey: .id)
            name = try container.decode(String.self, forKey: .name)
            avatarIconName = try container.decode(String.self, forKey: .avatarIconName)
            colorThemeName = try container.decode(String.self, forKey: .colorThemeName)
            sortOrder = try container.decode(Int.self, forKey: .sortOrder)
            ageYears = try container.decode(Int.self, forKey: .ageYears)
            dailyScreenTimeLimitMinutes = try container.decodeIfPresent(
